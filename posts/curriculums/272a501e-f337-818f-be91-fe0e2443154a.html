import flet as ft
from lib.models import todoGateway as gate
from lib.models.todoModel import Todo

dbname = 'sqlite3.db'

def main(page: ft.Page):
appbar = ft.AppBar(
center_title=True,
title=ft.Text(value="TODOアプリ"),
bgcolor=ft.colors.BLACK12
)

todo_list_control = ft.Column([])

def check_todo(e, isAchieved, id):
if isAchieved:
isAchieved = 0
else:
isAchieved = 1
gate.check_todo(isAchieved, id)
fetch_todo()

def fetch_todo():
rows = gate.get_todo()
todo_list = []
for row in rows:
todo = Todo(id=row[0],task=row[1],isAchieved=row[2] == 1,created_at=row[3])
todo_list.append(todo)
controls = []
for todo in todo_list:
text = ft.Text(todo.task)
checkbox = ft.Checkbox(on_change=lambda e, isAchieved=todo.isAchieved, id=todo.id:check_todo(e, isAchieved,id))
if todo.isAchieved:
text.style = ft.TextStyle(decoration=ft.TextDecoration.LINE_THROUGH)
checkbox.value = todo.isAchieved
controls.append(
ft.Row([
checkbox,
text,
ft.IconButton(icon=ft.icons.DELETE,on_click=lambda e,id=todo.id: delete_todo(e, id))
]))
todo_list_control.controls = controls
page.update()

def delete_todo(e, id):
gate.delete_todo(id)
fetch_todo()

def create_todo(e, task):
gate.save_todo(task)
page.go("/")

def route_change(e):
template_route = ft.TemplateRoute(page.route)

if template_route.match("/create"):
createTextField = ft.TextField(hint_text="TODO")
appbar.leading = ft.IconButton(icon=ft.icons.ARROW_BACK, on_click=lambda _:page.go("/"))
page.views.append(
ft.View(
"/create",
appbar=appbar,
controls=[
ft.Text("TODO作成"),
createTextField,
ft.ElevatedButton(text="作成",on_click=lambda e:create_todo(e, createTextField.value))
]
)
)
elif template_route.match("/"):
appbar.leading = ft.Text("")
floating_button = ft.FloatingActionButton(
icon=ft.icons.ADD,
bgcolor=ft.colors.BLUE_200,
on_click=lambda _: page.go("/create")
)
fetch_todo()
page.views.append(
ft.View(
"/",
appbar=appbar,
controls=[todo_list_control],
floating_action_button=floating_button
)
)
page.update()

page.on_route_change = route_change
page.go(page.route)

ft.app(main)

import sqlite3

dbname = 'sqlite3.db'

def get_todo():
conn = sqlite3.connect(dbname)
cur = conn.cursor()
cur.execute('SELECT * FROM todo')
rows = cur.fetchall()
conn.commit()
cur.close()
conn.close()
return rows

def save_todo(task):
conn = sqlite3.connect(dbname)
cur = conn.cursor()
cur.execute('INSERT INTO todo(task) VALUES (?)',(task,))
conn.commit()
cur.close()
conn.close()

def delete_todo(id):
conn = sqlite3.connect(dbname)
cur = conn.cursor()
cur.execute('DELETE FROM todo WHERE id = ?',(id,))
conn.commit()
cur.close()
conn.close()

def check_todo(isAchieved, id):
