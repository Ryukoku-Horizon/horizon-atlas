ページ移動可能にする

次に、ページ移動を可能にさせましょう。

main.py

python
1import flet as ft
2
3def main(page: ft.Page):
4    appbar = ft.AppBar(
5        center_title=True,
6        title=ft.Text(value="TODOアプリ"),
7        bgcolor=ft.Colors.BLACK12
8    )
9
10    def route_change(e):
11        template_route = ft.TemplateRoute(page.route)
12
13        if template_route.match("/create"):
14            appbar.leading = ft.IconButton(icon=ft.Icons.ARROW_BACK, on_click=lambda _:page.go("/"))
15            page.views.append(
16                ft.View(
17                    "/create",
18                    appbar=appbar,
19                    controls=[]
20                )
21            )
22        elif template_route.match("/"):
23            appbar.leading = ft.Text("")
24            floating_button = ft.FloatingActionButton(
25                icon=ft.Icons.ADD,
26                bgcolor=ft.Colors.BLUE_200,
27                on_click=lambda _: page.go("/create")
28            )
29            page.views.append(
30                ft.View(
31                    "/",
32                    appbar=appbar,
33                    controls=[],
34                    floating_action_button=floating_button
35                )
36            )
37        page.update()
38
39    page.on_route_change = route_change
40    page.go(page.route)
41
42ft.app(main)
43

a

ここではappbarを定義し、作成ページではappbarの左側にホームページに移動できるボタンを設置しました。

ホームページには、画面右下にfloating_buttonというボタンを設置し、作成ページに移動できるようにしました。
