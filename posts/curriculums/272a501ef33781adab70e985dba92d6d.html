db.py

python

import sqlite3

def init_db():
    dbname = 'sqlite3.db'
    conn = sqlite3.connect(dbname)
    cur = conn.cursor()
    cur.execute(
        'CREATE TABLE IF NOT EXISTS todo(id INTEGER PRIMARY KEY AUTOINCREMENT, task STRING, isAchieved BOOLEAN DEFAULT FALSE, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)')
    conn.commit()
    cur.close()
    conn.close()

import flet as ft
import sqlite3

dbname = 'sqlite3.db'

class Todo:
    def __init__(self, id=0, task='', isAchieved=False, created_at =''):
        self.id = id
        self.task = task
        self.isAchieved = isAchieved
        self.created_at = created_at

def fetch_todo():
	  conn = sqlite3.connect(dbname)
	  cur = conn.cursor()

	  todo_list = []
	  for row in rows:
	      todo = Todo(id=row[0],task=row[1],isAchieved=row[2] == 1,created_at=row[3])

	  controls = []
	  for todo in todo_list:
			  text = ft.Text(todo.task)
			  if todo.isAchieved:
            text.style = ft.TextStyle(decoration=ft.TextDecoration.LINE_THROUGH)
	      controls.append(

a

この関数では、まずデータベースからデータを全て取得し、そのデータをtodo_listというリストに格納します。次に、チェックボックスと、タスクを表示するTextコントロールと、削除ボタンを横に並べたRowコントロールを格納したcontrolsというリストを作成します。最後に、controlsを、todo_list_controlのcontrols属性の値として代入します。これでデータベースから取得したページに表示されます。

チェック機能

そして同じようにmain.py内のmain関数内でTodoが達成されたかチェックできる関数を作成します。

Todoをチェックする機能

python

def check_todo(e, isAchieved, id):
    if isAchieved:
        isAchieved = 0
    else:
        isAchieved = 1
    conn = sqlite3.connect(dbname)
    cur = conn.cursor()
    cur.execute('UPDATE todo SET isAchieved = ? WHERE id = ?',(isAchieved, id))
    conn.commit()
    cur.close()
    conn.close()
    fetch_todo()

a

isAchievedがTrueなら0（False）に切り替えるように、Falseなら1（True）に切り替えるようにしています。

def fetch_todo():
  conn = sqlite3.connect(dbname)
  cur = conn.cursor()
  cur.execute('SELECT * FROM todo')
  rows = cur.fetchall()
  todo_list = []
  for row in rows:
      todo = Todo(id=row[0],task=row[1],isAchieved=row[2] == 1,created_at=row[3])
      todo_list.append(todo)
  conn.commit()
  cur.close()
  conn.close()
  controls = []
  for todo in todo_list:
      text = ft.Text(todo.task)
      if todo.isAchieved:
          text.style = ft.TextStyle(decoration=ft.TextDecoration.LINE_THROUGH)
      controls.append(
          ft.Row([
		          # チェック機能実装

              text,
              # 削除機能実装
              ft.IconButton(icon=ft.icons.DELETE,on_click=lambda e,id=todo.id: delete_todo(e, id))
          ]))
  todo_list_control.controls = controls
